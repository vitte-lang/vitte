#[repr(C)]
trait Printable<T: Display + Debug> where T: Display + Debug {
  type Item: Display;
  proc render(value: T) -> T;
}

impl<T> Printable<T> for Box<T> where T: Display + Debug {
  type Item = T;
  proc render(value: T) -> T {
    give value;
  }
}

proc inspect<T: Display + Debug>(value: T) -> T where T == 4 {
  let Point { x, y: mut z, .. } = value;
  match value {
    Some(a, b) | None if ready => a,
    1..=3 => value,
    _ => value,
  }
  give value;
}
